\excercise 
By using Mehler's formula

$$
\frac{1}{\sqrt{1-\xi^2}}\exp\left[-\frac{x^2+y^2-2xy\xi}{\sqrt{1-\xi^2}}\right]=
e^{-(x^2+y^2)}\sum_n\frac{\xi^n}{2^nn!}H_n(x)H_n(y),
$$


\noindent prove that the propagator in the case of the one dimensional harmonic oscillator can be expressed as 	

\begin{align}
\notag K(x,t;x_0,t_0)=\sqrt{\frac{m\omega}{2\pi i\hbar\sin\omega(t-t_0)}} ~~~~~\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad~~~~~~~~~~~~~~~~~~\\
\times\exp\left\{\left(\frac{im\omega}{2\hbar\sin\omega (t-t_0)}\right) [(x^2+x_0^2)\cos\omega (t-t_0)-2xx_0]\right\}.~~~~~~~~~~~~~~~~~~
\end{align}

(\textbf{Hint:} use 

\begin{align}
K(x,t;x',t')=\sum_\alpha\bra{x}e^{-\frac{i}{\hbar}\mathbf{H}(t-t')}\ket{\alpha}\braket{\alpha}{x'}=\sum_\alpha\bra{x}e^{-\frac{i}{\hbar}E_\alpha(t-t')}\ket{\alpha}\braket{\alpha}{x'}.
\end{align}


\noindent for $ \ket{\alpha}=\ket{n} $.),

\excercise 

Derive explicitly the propagator in the p-representation, i.e.
 
$$
\bra{x}e^{-\frac{i}{\hbar}\mathbf{H}(t-t')}\ket{x'}=\int\mathcal{D}x\mathcal{D}p
\exp\left(\frac{i}{\hbar}\int_{t'}^t\bar{L}(\tau)d\tau\right),		
$$

\noindent where                        

\begin{align*}
\mathcal{D}x\mathcal{D}p=\lim_{\epsilon\rightarrow 0}\prod_n\left(\frac{dx_ndp_n}{2\pi\hbar}\right),~~~~~\int_{t'}^{t}\bar{L}(\tau)d\tau=\lim_{\epsilon\rightarrow 0}\epsilon\bar{L}_n,
\end{align*}

\noindent with

\begin{align*}
\bar{L}_n&=p_n\dot{x}_n-H(x_n,p_n),\\
\dot{x}_n&=\frac{1}{\epsilon}(x_n-x_{n-1}).
\end{align*}
