function calculate(){
var speedc =  2.99792458; // m/sec   
var x1GridSize = parseInt(document.getElementById("x1GridSize").value);
var x2GridSize = parseInt(document.getElementById("x2GridSize").value);
var nodeX1 = parseInt(document.getElementById("nodeX1").value);
var nodeX2 = parseInt(document.getElementById("nodeX2").value);
var ppc = parseInt(document.getElementById("ppc").value);
var totalpic = x1GridSize*x2GridSize*ppc;
var rawDataSizeByte = 7*4*totalpic;
var rawDataSizeMByte = 7*4*totalpic/1024/1024;
var rawDataSizeGByte = rawDataSizeMByte/1024;
var laserWavelength = parseFloat(document.getElementById("laserWavelength").value);
var omega0 = speedc*2*Math.PI/laserWavelength*1e5 //THz
var n0dens = 1.11/laserWavelength/laserWavelength; //unit 1/cm
n0dens = n0dens*1e27; //unit 1/cm
var dt = 1e3/omega0 ;// Unit. fs
var dL = laserWavelength/Math.PI/2; //nm
var xmin1 = parseFloat(document.getElementById("xmin1").value);
var xmin2 = parseFloat(document.getElementById("xmin2").value);
var xmax1 = parseFloat(document.getElementById("xmax1").value);
var xmax2 = parseFloat(document.getElementById("xmax2").value);
var gridX1Length = (xmax1 - xmin1)/ x1GridSize *1e3; // Unit nm
var gridX2Length = (xmax2 - xmin2)/ x2GridSize *1e3; // Unit nm
var gridX1Length2 = gridX1Length*gridX1Length;
var gridX2Length2 = gridX2Length*gridX2Length;
var cldL = Math.sqrt((gridX1Length2*gridX2Length2)/(gridX1Length2+gridX2Length2)); // nm
var cldt = cldL*1/speedc*1e-2; //fs
var dtFactor = parseFloat(document.getElementById("dtFactor").value);
var dtSimu = dtFactor*cldt ; 
var dtOSI = dtSimu/dt ;
var laserIntensity = parseFloat(document.getElementById("laserIntensity").value);
var a0 = 0.855*1e-12*Math.sqrt(laserIntensity)*laserWavelength ;
var TEField = 3.21*a0/laserWavelength*1e3 ;
document.getElementById("rawDataSizeMByte").value =  rawDataSizeMByte.toPrecision(3);
document.getElementById("rawDataSizeGByte").value =  rawDataSizeGByte.toPrecision(3); 
document.getElementById("totalpic").value =  totalpic;
document.getElementById("n0dens").value = n0dens.toExponential(3);//n0dens.toPrecision(3);
document.getElementById("omega0").value = omega0.toExponential(3);
document.getElementById("dt").value = dt.toPrecision(3);
document.getElementById("dL").value = dL.toPrecision(3);
document.getElementById("gridX1Length").value = gridX1Length.toPrecision(3);
document.getElementById("gridX2Length").value = gridX2Length.toPrecision(3);
document.getElementById("cldL").value = cldL.toPrecision(4);
document.getElementById("cldt").value = cldt.toPrecision(4);
document.getElementById("dtSimu").value = dtSimu.toPrecision(4);
document.getElementById("nodeX1OSI").value = nodeX1;
document.getElementById("nodeX2OSI").value = nodeX2;
document.getElementById("a0").value = a0.toPrecision(4);
document.getElementById("TEField").value = TEField.toPrecision(4);
//-----check for if_periodic in node_conf
if (parseInt(document.getElementById("ifPeriodX1").value)){
 document.getElementById("ifPeriodX1OSI").value = ".true."   
} else {
 document.getElementById("ifPeriodX1OSI").value = ".false."      
}
if (parseInt(document.getElementById("ifPeriodX2").value)){
 document.getElementById("ifPeriodX1OSI").value = ".true."   
} else {
 document.getElementById("ifPeriodX1OSI").value = ".false."      
}
//export simulation
document.getElementById("n0densOSI").value = n0dens.toExponential(4);//n0dens.toPrecision(3);
//export grid size
document.getElementById("x1GridSizeOSI").value = x1GridSize;
document.getElementById("x2GridSizeOSI").value = x2GridSize;
document.getElementById("dtOSI").value = dtOSI.toPrecision(4);;
//export zpulse
document.getElementById("a0OSI").value = a0.toPrecision(4);
}//function end
